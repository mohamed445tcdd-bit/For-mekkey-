const text =
"أنا مش عامل الصفحة دي عشان أقول كلمتين وخلاص… أنا عاملها عشان أقولك إنك لسه فارقة معايا، ولسه نفسي نرجع نضحك ونتكلم زي زمان.";

let textIndex = 0;


function typeText(){

    if(textIndex < text.length){

        document.getElementById(
            "typingText"
        ).innerHTML +=
            text.charAt(textIndex);

        textIndex++;

        setTimeout(
            typeText,
            35
        );

    }

}



/* ======================================
   ENTER SITE + MUSIC
====================================== */

function enterSite(){

    const opening =
        document.getElementById(
            "opening"
        );

    opening.classList.add(
        "hide"
    );


    /*
       تشغيل الأغنية هنا.

       لأنها نتيجة ضغطة زر،
       المتصفح يسمح بتشغيلها.
    */

    const song =
        document.getElementById(
            "song"
        );

    song.volume = 0.35;

    song.play().catch(
        function(){
            console.log(
                "Browser blocked audio."
            );
        }
    );


    typeText();

}

/* ======================================
   RUNNING YES BUTTON
====================================== */

const yesBtn =
    document.getElementById(
        "yesBtn"
    );

let attempts = 0;


const messages = [

    "استنى كده 😂",

    "مش بالسهولة دي يا قطتي 🐱",

    "هتوافقي يعني؟ 👀",

    "طب استني أفكر مكانك 😂🎀",

    "خلاص خلاص… اضغطي بقى ❤️"

];


function dodgeButton(){

    if(attempts >= 5){
        return;
    }


    attempts++;


    const x =
        (Math.random() * 260) - 130;


    const y =
        (Math.random() * 110) - 55;


    const rotate =
        (Math.random() * 14) - 7;


    yesBtn.style.transform =
        `translate(${x}px,${y}px)
         rotate(${rotate}deg)`;


    yesBtn.innerText =
        messages[
            attempts - 1
        ];


    if(attempts === 5){

        setTimeout(
            function(){

                yesBtn.style.transform =
                    "translate(0,0)";

                yesBtn.innerText =
                    "خلاص اضغطي بقى ❤️";

            },
            700
        );

    }

}



/* DESKTOP */

yesBtn.addEventListener(
    "mouseenter",
    dodgeButton
);



/* MOBILE */

yesBtn.addEventListener(
    "touchstart",
    function(e){

        if(attempts < 5){

            e.preventDefault();

            dodgeButton();

        }

    },
    {
        passive:false
    }
);



/* ======================================

    }

/* ======================================
   HEARTS
====================================== */

function createHearts(number){

    const emojis = [

        "💗",
        "💕",
        "🎀",
        "🐾",
        "✨",
        "❤️",
        "🌸"

    ];


    for(
        let i = 0;
        i < number;
        i++
    ){

        const heart =
            document.createElement(
                "div"
            );


        heart.className =
            "heart";


        heart.innerText =
            emojis[
                Math.floor(
                    Math.random()
                    * emojis.length
                )
            ];


        heart.style.left =
            Math.random()
            * 100
            + "vw";


        heart.style.fontSize =
            (15 +
            Math.random()*22)
            + "px";


        heart.style.animationDuration =
            (3 +
            Math.random()*4)
            + "s";


        document.body.appendChild(
            heart
        );


        setTimeout(
            function(){

                heart.remove();

            },
            7000
        );

    }

}
